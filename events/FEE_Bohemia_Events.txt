 	}

	}

			amount = 3
			adm = 30
			dip = 30
			mil = 30
		}

			name = boh_coinage_pioneer
			duration = 7300
		}
		custom_tooltip = fee_boh_thaler_princes_tt
		hidden_effect = {
			every_country = {
				limit = {
					is_part_of_hre = yes
				}
				add_country_modifier = {
					name = fee_boh_thaler
					duration = 7300
				}
			}
		}
	}

		name = FEE_Bohemia_Events.11.OPT2 #Yes yes, very nice, but what's in it for me?

			duration = 7300
		}
	}
}
