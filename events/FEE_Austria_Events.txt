#The Diet of Worms
country_event = {
	id = FEE_Austria_Events.1
	title = FEE_Austria_Events.1.T
	desc = FEE_Austria_Events.1.D
	picture = HRE_eventPicture

	fire_only_once = yes

	major = yes

	trigger = {
		is_emperor = yes
		is_year = 1450
		imperial_influence = 20
	}

	mean_time_to_happen = {
		months = 120
	}

	option = {
		name = FEE_Austria_Events.1.OPT1

		add_imperial_influence = 10
		if = {
			limit = {
				61 = {
					is_part_of_hre = yes
				}
			}
			61 = {
				add_base_tax = 1
			}
		}
		add_country_modifier = {
			name = fee_hab_reforms_diet_worms
			duration = 3650
		}
	}

	option = {
		name = FEE_Austria_Events.1.OPT2

		add_prestige_or_monarch_power = { amount = 10 }
		every_country = {
			limit = {
				is_part_of_hre = yes
			}
			add_opinion = {
				who = HAB
				modifier = fee_hab_protected_imperial_status_quo
			}
		}
	}
}

#Rise of the Ottomans in Europe
country_event = {
	id = FEE_Austria_Events.2
	title = FEE_Austria_Events.2.T
	desc = FEE_Austria_Events.2.D
	picture = HRE_eventPicture

	fire_only_once = yes

	major = yes

	trigger = {
		is_part_of_hre = yes
		is_elector = yes
		BYZ = {
			num_of_cities = 15
		}
		capital_scope = {
			continent = europe
		}
		OR = {
			any_owned_province = {
				any_neighbor_province = {
					owned_by = BYZ
				}
			}
			any_subject_country = {
				any_owned_province = {
					any_neighbor_province = {
						owned_by = BYZ
					}
				}
			}
		}
	}

	mean_time_to_happen = {
		months = 6
	}

	option = {
		name = FEE_Austria_Events.2.OPT1
		ai_chance = {
			factor = 100
		}

		add_mil_power = 50
		add_army_tradition_or_mil_power = { amount = 5 }
		add_opinion = {
			who = BYZ
			modifier = fee_hab_bulwark_christendom
		}
		reverse_add_opinion = {
			who = TUR
			modifier = fee_hab_bulwark_christendom
		}
	}
	option = {
		name = FEE_Austria_Events.2.OPT2
		ai_chance = {
			factor = 0
		}
		
		add_prestige_or_monarch_power = { amount = 10 }
		add_dip_power = 100
	}
}

#The Austrian Baroque
country_event = {
	id = FEE_Austria_Events.3
	title = FEE_Austria_Events.3.T
	desc = FEE_Austria_Events.3.D
	picture = COURT_eventPicture

	fire_only_once = yes

	trigger = {
		any_owned_province = {
			or = {
				area = inner_austria_area
				area = austria_proper_area
			}
		}
		is_year = 1650
		NOT = { is_year = 1730 }
		num_of_cities = 10
	}

	mean_time_to_happen = {
		months = 600
		modifier = {
			factor = 0.8
			stability = 2
		}
		modifier = {
			factor = 0.8
			legitimacy_equivalent = 100
		}
	}

	option = {
		name = FEE_Austria_Events.3.OPT1
		ai_chance = {
			factor = 85
		}

		add_country_modifier = {
			name = fee_hab_austrian_baroque
			duration = -1
		}
		custom_tooltip = fee_hab_austrian_continue_develop_tt
	}
	option = {
		name = FEE_Austria_Events.3.OPT2
		ai_chance = {
			factor = 15
		}

		add_stability_or_adm_power = yes
		add_adm_power = 10
		add_dip_power = 10
		add_mil_power = 10
	}
}
