#Province effect - used for buildings
update_improved_military_buildings_modifier = {
	if = {
		limit = {
			owner = { has_government_attribute = enables_military_exports }
		}
		hidden_effect = {
			remove_province_modifier = BM_military_exports_barracks_modifier
			remove_province_modifier = BM_military_exports_training_fields_modifier
			remove_province_modifier = BM_military_exports_regimental_camp_modifier
			remove_province_modifier = BM_military_exports_conscription_center_modifier
			remove_province_modifier = BM_military_exports_soldiers_households_modifier
			if = {
				limit = { has_building = barracks }
				add_province_modifier = {
					name = BM_military_exports_barracks_modifier
					duration = -1
				}
			}
			if = {
				limit = { has_building = training_fields }
				add_province_modifier = {
					name = BM_military_exports_training_fields_modifier
					duration = -1
				}
			}
			if = {
				limit = { has_building = regimental_camp }
				add_province_modifier = {
					name = BM_military_exports_regimental_camp_modifier
					duration = -1
				}
			}
			if = {
				limit = { has_building = conscription_center }
				add_province_modifier = {
					name = BM_military_exports_conscription_center_modifier
					duration = -1
				}
			}
			if = {
				limit = { has_building = soldier_households }
				add_province_modifier = {
					name = BM_military_exports_soldiers_households_modifier
					duration = -1
				}
			}
		}
	}
	if = {
		limit = {
			owner = { has_government_attribute = enable_portugese_marines }
		}
		hidden_effect = {
			remove_province_modifier = BM_portugese_marines_barracks_modifier
			remove_province_modifier = BM_portugese_marines_training_fields_modifier
			if = {
				limit = { has_building = barracks }
				add_province_modifier = {
					name = BM_portugese_marines_barracks_modifier
					duration = -1
				}
			}
			if = {
				limit = { has_building = training_fields }
				add_province_modifier = {
					name = BM_portugese_marines_training_fields_modifier
					duration = -1
				}
			}
		}
	}
}
