hessian_military_complex_reform = {
	icon = hesse_military
	allow_normal_conversion = yes
	valid_for_nation_designer = yes
	nation_designer_cost = 0
	potential = {
		OR = {
			tag = HES
			has_reform = hessian_militarized_monarchy_reform
			has_reform = hessian_military_complex_reform
		}
	}
	trigger = {
		OR = {
			tag = HES
			has_reform = hessian_militarized_monarchy_reform
			has_reform = hessian_military_complex_reform
		}
	}
	modifiers = {
		development_cost = -0.05
		state_maintenance_modifier = -0.25
	}
	custom_attributes = {
		enables_military_exports = yes #Barracks, Regimental Camps and Soldier Households are buffed 
	}
	effect = {
		hidden_effect = {
			every_owned_province = {
				if = {
					limit = { has_building = barracks }
					add_province_modifier = {
						name = BM_military_exports_barracks_modifier
						duration = -1
					}
				}
				if = {
					limit = { has_building = training_fields }
					add_province_modifier = {
						name = BM_military_exports_training_fields_modifier
						duration = -1
					}
				}
				if = {
					limit = { has_building = regimental_camp }
					add_province_modifier = {
						name = BM_military_exports_regimental_camp_modifier
						duration = -1
					}
				}
				if = {
					limit = { has_building = conscription_center }
					add_province_modifier = {
						name = BM_military_exports_conscription_center_modifier
						duration = -1
					}
				}
				if = {
					limit = { has_building = soldier_households }
					add_province_modifier = {
						name = BM_military_exports_soldiers_households_modifier
						duration = -1
					}
				}
			}
		}
	}
	removed_effect = {
		hidden_effect = {
			every_owned_province = {
				limit = {
					OR = {
						has_province_modifier = BM_military_exports_barracks_modifier
						has_province_modifier = BM_military_exports_training_fields_modifier
						has_province_modifier = BM_military_exports_regimental_camp_modifier
						has_province_modifier = BM_military_exports_conscription_center_modifier
						has_province_modifier = BM_military_exports_soldiers_households_modifier
					}
				}
				remove_province_modifier = BM_military_exports_barracks_modifier
				remove_province_modifier = BM_military_exports_training_fields_modifier
				remove_province_modifier = BM_military_exports_regimental_camp_modifier
				remove_province_modifier = BM_military_exports_conscription_center_modifier
				remove_province_modifier = BM_military_exports_soldiers_households_modifier
			}
		}
	}
	ai = {
		factor = 100
	}
}
