slave_farmer_reform = {
	icon = west_africa_slaves
	allow_normal_conversion = yes
	potential = {
		OR = {
			culture_group = mande
			has_reform = slave_farmer_reform
		}
	}
	trigger = {
		OR = {
			culture_group = mande
			has_reform = slave_farmer_reform
		}
	}
	modifiers = {
		manpower_recovery_speed = 0.05
		global_rebel_suppression_efficiency = 0.33
	}
	effect = {
		hidden_effect = {
			every_owned_province = {
				if = {
					limit = { 
						OR = {
							trade_goods = grain
							trade_goods = livestock
							trade_goods = slaves
						}
					}
					add_province_modifier = {
						name = BM_slave_settlements_modifier
						duration = -1
					}
				}
			}
		}
	}
	removed_effect = {
		hidden_effect = {
			every_owned_province = {
				limit = {
					has_province_modifier = BM_slave_settlements_modifier
				}
				remove_province_modifier = BM_slave_settlements_modifier
			}
		}
	}
	custom_attributes = {
		BM_enable_slave_settlements = yes
	}
	ai = {
		factor = 100
	}
}
